Class / Interface / DI
class Product
{
    public string Name;
    public decimal Price;

    public Product(string name, decimal price)
    {
        Name = name; Price = price;
    }

    public void ApplyDiscount(decimal pct)
        => Price -= Price * pct / 100;
}

interface IRepo { Order? GetById(int id); }

class OrderService
{
    private readonly IRepo _repo;
    public OrderService(IRepo repo) => _repo = repo;
}

คำสั่ง dotnet CLI — สร้างโปรเจกต์
dotnet --version              # เช็คเวอร์ชัน SDK
dotnet new webapp -o Name     # Razor Pages
dotnet new webapi -o Name     # Web API เปล่า

dotnet new sln                # สร้างไฟล์ solution (.sln)
dotnet sln add ./ProjName     # เพิ่ม project เข้า solution
คำสั่ง dotnet CLI — รันและ build
dotnet run                    # รันครั้งเดียว
dotnet watch run               # รัน + reload อัตโนมัติเมื่อแก้โค้ด
dotnet build                   # compile อย่างเดียว ไม่รัน
dotnet publish -c Release -o ./publish
dotnet test                    # รัน unit test ทั้งหมดในโปรเจกต์
คำสั่ง dotnet CLI — จัดการ package
dotnet add package ชื่อ Package
dotnet add package ชื่อ --version 8.0.0
dotnet remove package ชื่อ
dotnet list package             # ดู package ที่ติดตั้งไว้ทั้งหมด
dotnet list package --outdated  # เช็คว่า package ไหนมีเวอร์ชันใหม่กว่า
คำสั่ง dotnet CLI — EF Core (ฐานข้อมูล)
dotnet tool install --global dotnet-ef
dotnet tool update --global dotnet-ef

dotnet ef migrations add ชื่อ        # สร้าง migration ใหม่
dotnet ef migrations remove          # ลบ migration ล่าสุดที่ยังไม่ update
dotnet ef migrations list            # ดู migration ทั้งหมด
dotnet ef database update            # apply migration ล่าสุดเข้า database จริง
dotnet ef database update ชื่อ_migration_เก่า  # ย้อนกลับไป migration นั้น
dotnet ef database drop              # ลบ database ทิ้งทั้งหมด (ระวัง!)
dotnet ef dbcontext info             # ดูข้อมูล DbContext ที่ใช้อยู่

    {
        if (!ModelState.IsValid) return Page();
        return RedirectToPage("Index");
    }

}
EF Core — DbContext
public class AppDbContext : DbContext
{
    public AppDbContext(DbContextOptions<AppDbContext> o) : base(o) { }
    public DbSet<Product> Products => Set<Product>();
    public DbSet<Order> Orders => Set<Order>();
}

    public int CustomerId { get; set; }

Program.cs โครงมาตรฐาน
var builder = WebApplication.CreateBuilder(args);

builder.Services.AddRazorPages();
builder.Services.AddDbContext<AppDbContext>(o =>
    o.UseSqlite("Data Source=app.db"));
builder.Services.AddScoped<IRepo, Repo>();

var app = builder.Build();

app.UseHttpsRedirection();
app.UseAuthentication();   // ต้องมาก่อน UseAuthorization
app.UseAuthorization();
app.MapRazorPages();

app.Run();
Attribute ที่ใช้บ่อยใน Web API Controller
Attributeใช้ทำอะไร
[ApiController]ประกาศ class เป็น API controller
[Route("api/x")]กำหนด URL prefix
[HttpGet("{id}")]รับ GET พร้อม parameter จาก URL
[FromQuery]รับค่าจาก query string
[FromBody]รับค่าจาก JSON body
[Authorize]ต้อง login ก่อนถึงเข้าได้
Visual Studio (GUI) เทียบกับ VS Code (คำสั่ง/วิธีทำ)
ใน Visual Studio ทำผ่าน GUIใน VS Code ทำแบบนี้แทน
File → New → Projectterminal: dotnet new webapp -o ชื่อ (หรือ Command Palette → ".NET: New Project")
Solution Explorer → คลิกขวา → Add → New Item → Razor Pageสร้างไฟล์ .cshtml + .cshtml.cs เองในโฟลเดอร์ Pages/ (ไม่มี wizard ให้ แต่ VS Code auto-suggest boilerplate ให้บางส่วน)
Solution Explorer → คลิกขวา → Add → Classสร้างไฟล์ .cs เปล่าเอง แล้วพิมพ์ class ชื่อ
Tools → NuGet Package Manager → Manage NuGet Packagesterminal: dotnet add package ชื่อ หรือติดตั้ง extension "NuGet Gallery" เพิ่มถ้าอยากได้ UI คลิกเลือก
Package Manager Console: Add-Migration ชื่อterminal: dotnet ef migrations add ชื่อ (คำสั่งคนละชุด แต่ทำงานเหมือนกัน)

SQL Server Object Explorer (ดูข้อมูลใน table)ติดตั้ง extension "SQL Server (mssql)" ของ Microsoft แล้วเชื่อมต่อผ่านนั้น
Scaffolding: คลิกขวา → Add → New Scaffolded Item → Razor Pages using EFไม่มี wizard เทียบเท่า ต้องเขียนไฟล์เองตามที่สอนใน Module 15 (Full Project) — หรือติดตั้ง dotnet-aspnet-codegenerator เป็น global tool แล้วรันคำสั่ง scaffold เองผ่าน terminal (ใช้ยากกว่า เขียนเองมักเร็วกว่าสำหรับหน้าไม่กี่หน้า)
