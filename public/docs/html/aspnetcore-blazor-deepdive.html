Blazor เจาะลึก — สร้างระบบ CRUD ทีละขั้น

ใช่ครับ งานเว็บแอปฝั่ง ASP.NET ส่วนใหญ่ตอนนี้ไปทาง Blazor เป็นหลัก (แทนที่ Razor Pages แบบเดิมในหลายบริษัท) ไฟล์นี้เจาะจง Blazor ล้วนๆ สร้างระบบจริงทีละขั้น เริ่มจาก CRUD ง่ายที่สุด แล้วค่อยๆ เพิ่มฟีเจอร์ ทุกขั้นบอกชื่อไฟล์ ชื่อโปรเจกต์ และคำสั่ง CLI ที่ต้องพิมพ์ใน VS Code ชัดเจนเสมอ

ชื่อโปรเจกต์ที่ใช้ตลอดไฟล์นี้: BlazorCrudApp — ถ้าตั้งชื่ออื่น ให้แทนที่คำนี้ทุกจุดที่เจอในโค้ดตัวอย่าง (namespace จะอิงชื่อนี้ด้วย)

using System.ComponentModel.DataAnnotations;

namespace BlazorCrudApp.Models;

public class Product
{
    public int Id { get; set; }

    [Required(ErrorMessage = "กรุณากรอกชื่อสินค้า")]

    public string Name { get; set; } = string.Empty;

    [Range(0, 1000000)]
    public decimal Price { get; set; }

    public int StockQuantity { get; set; }

    public string Category { get; set; } = "ทั่วไป";
}

@page "/"
@rendermode InteractiveServer
@using BlazorCrudApp.Data
@using BlazorCrudApp.Models
@using Microsoft.EntityFrameworkCore
@inject AppDbContext Db

<PageTitle>รายการสินค้า</PageTitle>

<h1>รายการสินค้า</h1>

<a href="/create">+ เพิ่มสินค้าใหม่</a>

<table class="table">
    <thead>
        <tr><th>ชื่อ</th><th>ราคา</th><th>คงเหลือ</th><th></th></tr>
    </thead>
    <tbody>
    @foreach (var product in products)
    {
        <tr>
            <td>@product.Name</td>
            <td>@product.Price บาท</td>
            <td>@product.StockQuantity</td>
            <td>
                <a href="@($"/edit/{product.Id}")">แก้ไข</a>
                <button @onclick="() => DeleteAsync(product.Id)">ลบ</button>
            </td>
        </tr>
    }
    </tbody>
</table>

@code {
    private List<Product> products = new();

    protected override async Task OnInitializedAsync()
    {
        products = await Db.Products.AsNoTracking().ToListAsync();
    }

    private async Task DeleteAsync(int id)
    {
        var product = await Db.Products.FindAsync(id);
        if (product is not null)
        {
            Db.Products.Remove(product);
            await Db.SaveChangesAsync();
            products = await Db.Products.AsNoTracking().ToListAsync();
        }
    }
}

ความหมายของสิ่งที่ต่างจาก Razor Pages ที่เคยเรียน:

@rendermode InteractiveServer — บังคับให้หน้านี้ interactive (ไม่งั้นปุ่มลบจะกดไม่ได้ ตามที่เตือนไว้ใน Module 6 ของ webapp-deepdive)

@inject AppDbContext Db — วิธี inject service เข้ามาใช้ในไฟล์ .razor โดยตรง (ไม่ต้องผ่าน constructor แบบ PageModel)

OnInitializedAsync — เทียบเท่า OnGetAsync ของ Razor Pages ทำงานตอน component โหลดครั้งแรก

ลองทำ: รันโปรเจกต์แล้วเปิดหน้าแรก ควรเห็นตารางว่างเปล่า (ยังไม่มีข้อมูล) ลองเพิ่มข้อมูลตรงๆ ผ่าน SQLite ก่อน หรือรอทำหน้า Create ใน Step ถัดไปแล้วค่อยทดสอบพร้อมกัน

จุดที่ต่างจากฟอร์มของ Razor Pages ชัดเจนที่สุด:

<EditForm Model="..." OnValidSubmit="..."> — คือ component ฟอร์มของ Blazor เอง แทนที่ <form method="post"> ธรรมดา ตรวจ validation ให้อัตโนมัติก่อนเรียก OnValidSubmit เสมอ (ถ้าข้อมูลไม่ผ่าน [Required]/[Range] จะไม่เรียก method นี้เลย)

<InputText @bind-Value="..."> — เทียบเท่า asp-for ของ Razor Pages แต่ผูกค่าแบบ two-way binding ทันที ไม่ต้อง submit ก่อนถึงจะเห็นค่าอัพเดท

NavigationManager — ตัวที่ใช้เปลี่ยนหน้าใน Blazor เทียบเท่า RedirectToPage() ของ Razor Pages

ลืมบ่อย: ลืม @rendermode InteractiveServer แล้วกดปุ่มบันทึกไม่ทำงานเลย (ฟอร์มดูปกติแต่กดแล้วไม่มีอะไรเกิดขึ้น เพราะหน้ายัง static อยู่)
ลองทำ: เพิ่มลิงก์ "แก้ไข" ในหน้า NavMenu.razor ให้ชี้มาที่ /create ด้วย แล้วทดสอบเพิ่มสินค้า 2-3 ชิ้น กลับไปหน้าแรกดูว่าแสดงครบไหม
STEP 6หน้า Update — แก้ไขข้อมูล

@page "/edit/{Id:int}"
@rendermode InteractiveServer
@using BlazorCrudApp.Data
@using BlazorCrudApp.Models
@inject AppDbContext Db
@inject NavigationManager Nav

<PageTitle>แก้ไขสินค้า</PageTitle>

<h1>แก้ไขสินค้า</h1>

@if (editProduct is not null)
{
    <EditForm Model="editProduct" OnValidSubmit="SaveAsync">
        <DataAnnotationsValidator />
        <ValidationSummary />

        <label>ชื่อสินค้า</label>
        <InputText @bind-Value="editProduct.Name" />

        <label>ราคา</label>
        <InputNumber @bind-Value="editProduct.Price" />

        <button type="submit">บันทึกการแก้ไข</button>
    </EditForm>
}
else
{
    <p>กำลังโหลด...</p>
}

@code {
    [Parameter] public int Id { get; set; }
    private Product? editProduct;

    protected override async Task OnInitializedAsync()
    {
        editProduct = await Db.Products.FindAsync(Id);
    }

    private async Task SaveAsync()
    {
        Db.Products.Update(editProduct!);
        await Db.SaveChangesAsync();
        Nav.NavigateTo("/");
    }
}

@page "/edit/{Id:int}" — รับค่าจาก URL แบบเดียวกับ @page "{id:int}" ของ Razor Pages แต่ตัวแปรต้องประกาศเป็น [Parameter] คู่กับชื่อใน route (ตัวพิมพ์ใหญ่-เล็ก ต้องตรงกัน — {Id:int} คู่กับ public int Id)

if (editProduct is not null) — จำเป็นมากเพราะ OnInitializedAsync เป็น async ตัวหน้าเว็บจะ render ก่อนข้อมูลโหลดเสร็จเสมอ ถ้าไม่เช็ค null จะ error ทันที (ตรงกับที่เตือนไว้ใน FAQ หมวด Blazor)

ลองทำ: ทดสอบกดปุ่ม "แก้ไข" จากหน้าแรก ตรวจว่าค่าเดิมโหลดมาแสดงในฟอร์มถูกต้องก่อน submit
STEP 7Delete — ลบข้อมูล
5 นาที
ทำไมสั้น: เขียนปุ่มลบไว้แล้วตั้งแต่ Step 4 (method DeleteAsync ในหน้า Home.razor) เพราะ Blazor ไม่ต้องมี handler พิเศษแบบ asp-page-handler="Delete" ของ Razor Pages — แค่ผูก @onclick เข้ากับ method ธรรมดาได้เลย ครบ CRUD แล้วตั้งแต่ Step 6

เช็คลิสต์ก่อนไปต่อ
เพิ่มสินค้าใหม่ผ่าน /create ได้ → เห็นในหน้าแรกทันที (Read) → กดแก้ไขแล้วค่าเดิมโหลดมาถูก → บันทึกแล้วค่าที่หน้าแรกอัพเดท → กดลบแล้วแถวหายไปจากตารางทันทีโดยไม่ reload หน้าเลย ถ้าทุกอย่างผ่าน แปลว่าพร้อมไปต่อฟีเจอร์เสริมแล้ว

STEP 9เพิ่ม Filter แบบ dropdown
15 นาที
แก้ไฟล์: Components/Pages/Home.razor
<select @bind="selectedCategory">
    <option value="">ทุกหมวดหมู่</option>
    @foreach (var cat in products.Select(p => p.Category).Distinct())
    {
        <option value="@cat">@cat</option>
    }
</select>

@code {
    private string selectedCategory = "";

    private IEnumerable<Product> FilteredProducts =>
        products
            .Where(p => p.Name.Contains(searchText, StringComparison.OrdinalIgnoreCase))
            .Where(p => string.IsNullOrEmpty(selectedCategory) || p.Category == selectedCategory);
}

เพิ่มเงื่อนไข .Where ต่อกันเรื่อยๆ ใน FilteredProducts ได้ตามต้องการ รูปแบบนี้ขยายไปเป็น filter หลายตัวพร้อมกันได้ (เหมือนที่ระบบสต๊อกจริงมักมี filter ปี/เดือน/หมวดพร้อมกัน)

STEP 10แบ่งหน้า (Pagination) เป็น Component แยก
25 นาที
ทำไมแยก component: pagination เป็นของที่ใช้ซ้ำได้ทุกหน้าตาราง แยกไว้ครั้งเดียวใช้ได้ตลอดโปรเจกต์
สร้างไฟล์ใหม่: Components/Shared/Pagination.razor
<div class="pagination">
    <button disabled="@(CurrentPage == 1)" @onclick="() => ChangePage(CurrentPage - 1)">ก่อนหน้า</button>

    <span>หน้า @CurrentPage จาก @TotalPages</span>

    <button disabled="@(CurrentPage == TotalPages)" @onclick="() => ChangePage(CurrentPage + 1)">ถัดไป</button>
</div>

@code {
    [Parameter] public int CurrentPage { get; set; }
    [Parameter] public int TotalPages { get; set; }
    [Parameter] public EventCallback<int> CurrentPageChanged { get; set; }

    private async Task ChangePage(int newPage)
    {
        if (newPage >= 1 && newPage <= TotalPages)
        {
            await CurrentPageChanged.InvokeAsync(newPage);
        }
    }
}
แก้ไฟล์: Components/Pages/Home.razor
<Pagination CurrentPage="currentPage" TotalPages="totalPages"
            CurrentPageChanged="@(page => currentPage = page)" />

@code {
    private int currentPage = 1;
    private const int pageSize = 10;

    private int totalPages =>
        (int)Math.Ceiling(FilteredProducts.Count() / (double)pageSize);

    private IEnumerable<Product> PagedProducts =>
        FilteredProducts.Skip((currentPage - 1) * pageSize).Take(pageSize);
}

แก้ @foreach ในตารางจาก FilteredProducts เป็น PagedProducts แทน [Parameter] public EventCallback<int> CurrentPageChanged คือรูปแบบมาตรฐานของ Blazor สำหรับทำ two-way binding เอง (ตั้งชื่อ XxxChanged คู่กับ parameter Xxx เสมอ ทำให้ใช้ @bind-CurrentPage="currentPage" แบบสั้นได้ในอนาคตถ้าต้องการ)

ลืมบ่อย: เปลี่ยน filter (เช่นพิมพ์ค้นหาใหม่) แล้วลืม reset currentPage กลับเป็น 1 ทำให้ค้างอยู่หน้า 5 ทั้งที่ผลลัพธ์กรองใหม่มีแค่ 2 หน้า ควรเพิ่มโค้ด reset currentPage=1 ทุกครั้งที่ searchText หรือ selectedCategory เปลี่ยน

    private void SortBy(string column)
    {
        if (sortColumn == column)
        {
            sortAscending = !sortAscending;
        }
        else
        {
            sortColumn = column;
            sortAscending = true;
        }
    }

    private IEnumerable<Product> SortedProducts
    {
        get
        {
            var sorted = sortColumn switch
            {
                "Price" => FilteredProducts.OrderBy(p => p.Price),
                _ => FilteredProducts.OrderBy(p => p.Name)
            };
            return sortAscending ? sorted : sorted.Reverse();
        }
    }
}

ต้องเปลี่ยนจุดที่ PagedProducts อ้างอิง FilteredProducts เดิม ให้ไปอ้างอิง SortedProducts แทน (เรียง ก่อน ตัดหน้าเสมอ ไม่งั้นแบ่งหน้าจะสลับมั่ว)

STEP 14เพิ่ม Chart ด้วย JS Interop
25 นาที
ทำไมต้องพึ่ง JavaScript: Blazor ไม่มี chart library ในตัว การวาดกราฟ (เช่น Chart.js) ต้องเรียกผ่าน JS interop เป็นหนึ่งในไม่กี่จุดที่ Blazor "ต้อง" แตะ JavaScript จริงๆ
สร้างไฟล์ใหม่: wwwroot/js/chartInterop.js
window.renderStockChart = (canvasId, labels, data) => {
    new Chart(document.getElementById(canvasId), {
        type: 'bar',
        data: {
            labels: labels,
            datasets: [{ label: 'จำนวนคงเหลือ', data: data }]
        }
    });
};
แก้ไฟล์: Components/App.razor (เพิ่ม script tag)

หาแท็ก </body> แล้วเพิ่มก่อนบรรทัดนั้น (ต้องโหลด Chart.js จาก CDN ก่อนไฟล์ของเราเอง):

<script src="https://cdn.jsdelivr.net/npm/chart.js"></script>
<script src="js/chartInterop.js"></script>
สร้างไฟล์ใหม่: Components/Pages/Dashboard.razor
@page "/dashboard"
@rendermode InteractiveServer
@inject AppDbContext Db
@inject IJSRuntime JS

<h1>Dashboard</h1>
<canvas id="stockChart"></canvas>

@code {
    protected override async Task OnAfterRenderAsync(bool firstRender)
    {
        if (firstRender)
        {
            var products = await Db.Products.AsNoTracking().ToListAsync();
            var labels = products.Select(p => p.Name).ToArray();
            var data = products.Select(p => p.StockQuantity).ToArray();

            await JS.InvokeVoidAsync("renderStockChart", "stockChart", labels, data);
        }
    }
}

OnAfterRenderAsync (ไม่ใช่ OnInitializedAsync) เพราะต้องรอให้ HTML element <canvas> ถูกวาดขึ้นจอจริงก่อน ถึงจะเรียก JS ไปวาดกราฟลงบน element นั้นได้ พารามิเตอร์ firstRender กันไม่ให้วาดกราฟซ้ำทุกครั้งที่หน้าจอ re-render

ลืมบ่อย: ลืมเพิ่ม @inject IJSRuntime JS หรือลืมโหลด Chart.js จาก CDN ก่อน ไฟล์ chartInterop.js ของเราเอง (ลำดับ script tag สำคัญ — ต้องโหลด Chart.js ก่อนเสมอ)
