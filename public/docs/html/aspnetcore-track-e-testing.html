MODULE 3Unit Test แรก
2 วัน

เริ่มจากทดสอบ Order.TotalAmount ที่เขียนไว้ใน Track A Module 3 (logic ล้วนๆ ไม่มี dependency ภายนอก จึงเหมาะเป็น test แรก)

สร้างไฟล์ใหม่: BlazorCrudApp.Tests/OrderTests.cs
using BlazorCrudApp.Models;
using Xunit;

public class OrderTests
{
    [Fact]
    public void TotalAmount_SumsAllItems()
    {
        // Arrange — เตรียมข้อมูล
        var order = new Order
        {
            Items = new List<OrderItem>
            {
                new() { Quantity = 2, PriceAtOrderTime = 100 },
                new() { Quantity = 1, PriceAtOrderTime = 50 }
            }
        };

        // Act — เรียกสิ่งที่ต้องการทดสอบ
        var total = order.TotalAmount;

        // Assert — ตรวจผลลัพธ์
        Assert.Equal(250, total);
    }

    [Fact]
    public void TotalAmount_EmptyOrder_ReturnsZero()
    {
        var order = new Order();
        Assert.Equal(0, order.TotalAmount);
    }

    [Fact]
    public void TotalAmount_UsesPriceAtOrderTime_NotCurrentProductPrice()
    {
        var order = new Order
        {
            Items = new List<OrderItem>
            {
                new()
                {
                    Quantity = 1,
                    PriceAtOrderTime = 100,
                    Product = new Product { CurrentPrice = 999 }
                }
            }
        };

        Assert.Equal(100, order.TotalAmount);
    }
}

รูปแบบ Arrange-Act-Assert คือโครงสร้างมาตรฐานของ test เกือบทุกอัน: เตรียมข้อมูล → เรียกสิ่งที่ทดสอบ → ตรวจผลลัพธ์ อ่านแล้วเข้าใจได้ทันทีโดยไม่ต้องมีคำอธิบายเพิ่ม

test ตัวที่ 3 คือตัวที่ "ป้องกันบั๊กจริง" จาก Track A Module 1 — ถ้าวันหนึ่งมีคนแก้ TotalAmount ไปใช้ CurrentPrice แทนโดยไม่ตั้งใจ test นี้จะพังทันทีตอนรัน ก่อนที่บั๊กจะไปถึงผู้ใช้จริง

MODULE 4Test ที่มี Dependency (Mock)
3 วัน
ปัญหา: OrderService (จาก Track A/Module 11 ของ webapp-deepdive) เรียก external API เช็คสต๊อก ไม่อยากให้ test ต้องต่อ API จริงทุกครั้งที่รัน (ช้า, ไม่เสถียร, อาจตัดสต๊อกจริงของบริษัท) → ใช้ "Mock" แทน — object ปลอมที่จำลองพฤติกรรมตามที่กำหนดเอง
สร้างไฟล์ใหม่: BlazorCrudApp.Tests/OrderServiceTests.cs
using Moq;
using Xunit;

public class OrderServiceTests
{
    [Fact]
    public async Task ConfirmOrder_InsufficientStock_ReturnsFailure()
    {
        // Arrange — สร้าง mock ของ dependency
        var mockStockCheck = new Mock<IStockCheckService>();
        mockStockCheck
            .Setup(s => s.GetStockAsync(It.IsAny<int>()))
            .ReturnsAsync(5);   // จำลองว่าสต๊อกเหลือแค่ 5

        var mockRepo = new Mock<IOrderRepository>();
        var service = new OrderService(mockRepo.Object, mockStockCheck.Object);

        // Act — สั่งซื้อ 10 ชิ้น (เกินสต๊อก)
        var result = await service.ConfirmOrderAsync(productId: 1, quantity: 10);

        // Assert
        Assert.False(result.Success);
        Assert.Contains("สต๊อกไม่พอ", result.ErrorMessage);

        // ยืนยันว่า "ไม่ได้บันทึกอะไรลงฐานข้อมูล" เพราะสต๊อกไม่พอ
        mockRepo.Verify(r => r.AddAsync(It.IsAny<Order>()), Times.Never);
    }
}

Setup(...).ReturnsAsync(5) คือการกำหนดว่า "ถ้าโดนเรียก method นี้ ให้ตอบกลับค่านี้" ส่วน Verify(..., Times.Never) คือการตรวจว่า "method นี้ต้องไม่ถูกเรียกเลย" ซึ่งสำคัญมากสำหรับเช็คว่า logic ป้องกันทำงานจริง (ไม่บันทึกออเดอร์เมื่อสต๊อกไม่พอ)

ข้อควรรู้: การ mock ได้ต้องออกแบบโค้ดให้รับ dependency ผ่าน interface (IStockCheckService, IOrderRepository) — นี่คือเหตุผลจริงที่ Track A/Module 12 ของ webapp-deepdive สอนให้แยก Service/Repository ผ่าน interface ตั้งแต่แรก ไม่ใช่แค่ทำให้โค้ดดูเป็นระเบียบ
MODULE 5Integration Test กับฐานข้อมูลจริง
3 วัน
เมื่อ Mock ไม่พอ: Mock ทดสอบ "logic ของเรา" ได้ดี แต่ไม่ได้ทดสอบว่า query EF Core ที่เขียนไว้แปลงเป็น SQL ถูกจริงไหม (เช่น Include ครบไหม, GroupBy ทำงานได้ไหม) ต้องใช้ database จริงทดสอบ
ใช้ SQLite In-Memory — database จริงแต่อยู่ใน RAM ลบทิ้งอัตโนมัติหลังจบ test
dotnet add package Microsoft.EntityFrameworkCore.Sqlite
สร้างไฟล์ใหม่: BlazorCrudApp.Tests/OrderQueryTests.cs
using Microsoft.Data.Sqlite;
using Microsoft.EntityFrameworkCore;

public class OrderQueryTests : IDisposable
{
    private readonly SqliteConnection _connection;
    private readonly AppDbContext _db;

    public OrderQueryTests()
    {
        _connection = new SqliteConnection("DataSource=:memory:");
        _connection.Open();

        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseSqlite(_connection).Options;

        _db = new AppDbContext(options);
        _db.Database.EnsureCreated();
    }

    [Fact]
    public async Task GetOrder_IncludesItemsAndProducts()
    {
        // Arrange — ใส่ข้อมูลทดสอบลงฐานข้อมูลจริง (ใน RAM)
        var product = new Product { Name = "ทดสอบ", CurrentPrice = 100 };
        var order = new Order
        {
            Items = new List<OrderItem>
            {
                new() { Product = product, Quantity = 2, PriceAtOrderTime = 100 }
            }
        };
        _db.Orders.Add(order);
        await _db.SaveChangesAsync();

        // Act
        var loaded = await _db.Orders
            .Include(o => o.Items).ThenInclude(i => i.Product)
            .FirstAsync();

        // Assert
        Assert.Single(loaded.Items);
        Assert.NotNull(loaded.Items[0].Product);
        Assert.Equal("ทดสอบ", loaded.Items[0].Product!.Name);
    }

    public void Dispose()
    {
        _db.Dispose();
        _connection.Dispose();
    }
}

DataSource=:memory: สร้างฐานข้อมูล SQLite ที่อยู่ใน RAM เท่านั้น หายไปทันทีที่ปิด connection — ทำให้แต่ละ test เริ่มจากฐานข้อมูลว่างเปล่าสะอาดเสมอ ไม่กระทบกันเอง และไม่แตะฐานข้อมูลจริงของโปรเจกต์เลย

MODULE 6เขียน Test ตอนเจอบั๊ก
2 วัน
นิสัยสำคัญที่ใช้จริงในทีม: ทุกครั้งที่เจอบั๊ก (จาก Track B) ให้เขียน test ที่ "ทำให้บั๊กนั้นเกิดซ้ำได้" ก่อนแก้ พอ test ล้มเหลวตรงตามบั๊กแล้วค่อยแก้โค้ดให้ test ผ่าน บั๊กเดิมจะไม่มีวันกลับมาโดยไม่มีใครรู้ เพราะมี test คอยเฝ้าอยู่
ตัวอย่างลำดับการทำงาน

1. ได้รับรายงานบั๊ก: "ออเดอร์ที่มีสินค้า 0 ชิ้นสามารถยืนยันได้"

2. เขียน test ที่ทำให้บั๊กเกิดซ้ำก่อน (ยังไม่แก้โค้ด):

[Fact]
public async Task ConfirmOrder_EmptyCart_ReturnsFailure()
{
    var result = await service.ConfirmOrderAsync(new List<CartLine>());
    Assert.False(result.Success);
}

3. รัน test → ล้มเหลว (พิสูจน์ว่าเจอบั๊กจริงตามที่รายงาน)

4. แก้โค้ดจนกว่า test ผ่าน

5. เก็บ test นี้ไว้ถาวร — เป็น "regression test" ที่ป้องกันบั๊กเดิมกลับมา

MODULE 7Test ใน Blazor Component (bUnit)
3 วัน
รันในโปรเจกต์ Test
dotnet add package bunit
สร้างไฟล์ใหม่: BlazorCrudApp.Tests/PaginationTests.cs
using Bunit;
using Xunit;

public class PaginationTests : TestContext
{
    [Fact]
    public void PreviousButton_DisabledOnFirstPage()
    {
        var cut = RenderComponent<Pagination>(parameters => parameters
            .Add(p => p.CurrentPage, 1)
            .Add(p => p.TotalPages, 5));

        var prevButton = cut.Find("button:first-child");
        Assert.True(prevButton.HasAttribute("disabled"));
    }

    [Fact]
    public void ClickNext_InvokesCallbackWithNextPage()
    {
        int? receivedPage = null;
        var cut = RenderComponent<Pagination>(parameters => parameters
            .Add(p => p.CurrentPage, 2)
            .Add(p => p.TotalPages, 5)
            .Add(p => p.CurrentPageChanged, page => receivedPage = page));

        cut.Find("button:last-child").Click();

        Assert.Equal(3, receivedPage);
    }
}

bUnit ให้ render component ใน memory (ไม่ต้องเปิด browser) แล้ว "คลิก" ปุ่มจำลองตรวจผลได้ เร็วกว่าและเสถียรกว่าทดสอบผ่านหน้าเว็บจริงมาก เหมาะกับ component ที่มี logic เยอะ เช่น Pagination จากบทเรียน Blazor deepdive
