ต่อยอด IWS Clone ด้วยความรู้จาก RPG จริง

เอา pattern จาก CHKASSY.SQLRPG และ generic JSON engine มาสร้างเป็น Node.js + pub400 จริง

ข่าวดี: "Generic JSON Engine" ที่ RPG ต้องเขียนโค้ดเยอะมาก (query QSYS2.SYSCOLUMNS เอง, dynamic SQL, PREPARE/EXECUTE) — ใน Node.js ได้มาฟรีอยู่แล้ว เพราะ odbc package คืนผลลัพธ์เป็น JS object ที่มีชื่อ column ตรงตัวอัตโนมัติ ไม่ต้อง introspect เอง สิ่งที่ต้องสร้างเลียนแบบจริงๆ คือสถาปัตยกรรมการแยกชั้น (business logic ↔ response formatting) ไม่ใช่เทคนิค column-introspection

2. Generic Query Runner (แทนส่วน error-handling ที่ RPG ทำซ้ำทุกที่)

เลียนแบบ pattern Monitor/On-Error ที่ RPG ใช้ครอบทุก SQL block — ทำเป็นฟังก์ชันกลางครั้งเดียว ใช้ซ้ำได้ทุก route

// lib/queryRunner.js
const { getConnection } = require('./db');

async function runQuery(sql, params = []) {
  let connection;
  try {
    connection = await getConnection();
    const result = await connection.query(sql, params);
    return { ok: true, data: result };
  } catch (err) {
    // เลียนแบบ pattern error JSON ที่เจอใน RPG:
    // '{"status": "error", "message": "..." }'
    return { ok: false, error: err.message };
  } finally {
    if (connection) await connection.close();
  }
}

module.exports = { runQuery };

3. จำลอง CHKASSY.SQLRPG — join chain เดียวกันเป๊ะ

แปลง logic จาก RPG (query ทีละตาราง, ถ้าไม่เจอให้เป็นค่าว่าง ไม่ throw error) เป็น Node.js ตรงๆ

// routes/talinf.js
const express = require('express');
const router = express.Router();
const { runQuery } = require('../lib/queryRunner');

router.get('/talinf', async (req, res) => {
  const lotNo = req.query.lotno;
  if (!lotNo) return res.status(400).json({ error: 'lotno is required' });

  // Step 1: หาใน talinf ด้วย CNASSY (ตรงกับ CHAIN/SELECT แรกใน CHKASSY)
  const main = await runQuery(
    'SELECT CDDEVC, CDWINP, CNMARK, CNWEFR, QTSTKP, QTSTKS FROM MYLIB.TALINF WHERE CNASSY = ?',
    [lotNo]
  );

  if (!main.ok || main.data.length === 0) {
    // ตรงกับ p_isFound = *off ใน RPG
    return res.json({ P_ISFOUND: "0" });
  }

  const row = main.data[0];
  let nmwefr = "";
  let nmdevc = "";
  let nmpakg = "";

  // Step 2: resolve wafer name (twlinf → mwafer) — ถ้าไม่เจอ ปล่อยว่างไว้ ไม่ error
  const wefr = await runQuery(
    'SELECT CDWEFR FROM MYLIB.TWLINF WHERE CNWEFR = ?', [row.CNWEFR]
  );
  if (wefr.ok && wefr.data.length > 0) {
    const wafer = await runQuery(
      'SELECT NMWEFR FROM MYLIB.MWAFER WHERE CDWEFR = ?', [wefr.data[0].CDWEFR]
    );
    if (wafer.ok && wafer.data.length > 0) nmwefr = wafer.data[0].NMWEFR;
  }

  // Step 3: resolve device + package name (mdvice → MDVGRP → mpackg)
  const dvice = await runQuery(
    'SELECT NMDEVC, CDDGRP FROM MYLIB.MDVICE WHERE CDDEVC = ?', [row.CDDEVC]
  );
  if (dvice.ok && dvice.data.length > 0) {
    nmdevc = dvice.data[0].NMDEVC;
    const dgrp = await runQuery(
      'SELECT CDPAKG FROM MYLIB.MDVGRP WHERE CDDGRP = ?', [dvice.data[0].CDDGRP]
    );
    if (dgrp.ok && dgrp.data.length > 0) {
      const pakg = await runQuery(
        'SELECT NMPAKG FROM MYLIB.MPACKG WHERE CDPAKG = ?', [dgrp.data[0].CDPAKG]
      );
      if (pakg.ok && pakg.data.length > 0) nmpakg = pakg.data[0].NMPAKG;
    }
  }

  // Mapping layer — จำลองสิ่งที่ IWS ทำอัตโนมัติตอน deploy
  res.json({
    P_ISFOUND: "1",
    DEVICE_CODE: row.CDDEVC,
    WIP_CODE: row.CDWINP,
    MARK_NO: row.CNMARK,
    WAFER_LOT_NO: row.CNWEFR,
    QTY_PCS: row.QTSTKP,
    QTY_SEC: row.QTSTKS,
    DEVICE_NAME: nmdevc,
    PACKAGE_NAME: nmpakg,
    WAFER_NAME: nmwefr,
  });
});

module.exports = router;
เทียบกับ RPG ต้นฉบับ: RPG ใช้ if sqlcode = 0 ... else ... endif ซ้อนกันหลายชั้น — Node.js ใช้ if (x.ok && x.data.length > 0) pattern เดียวกันเป๊ะ แค่ syntax คนละภาษา logic เหมือนกัน 100%
