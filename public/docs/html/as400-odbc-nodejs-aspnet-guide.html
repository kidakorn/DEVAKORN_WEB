require('dotenv').config();

// Fail fast: a missing setting should stop the app at startup with a clear
// message, not surface later as a confusing ODBC error on the first request.
const required = ['AS400_DRIVER', 'AS400_HOST', 'AS400_USER', 'AS400_PASSWORD', 'AS400_LIBRARY'];

}

1.4 db.js: connection pool แบบมีชั้นป้องกัน

db.js
'use strict';
const odbc = require('odbc');
const cfg = require('./config');

// ODBC rule: wrap a value in braces and double any closing brace.
// Why: without this, a password containing ';' or '}' silently splits the
// connection string and produces a misleading "login failed" error.
const braced = (value) => `{${String(value).replace(/}/g, '}}')}`;

function buildConnectionString() {
  const parts = [
    `DRIVER=${braced(cfg.driver)}`,
    `SYSTEM=${cfg.host}`,
    `UID=${braced(cfg.user)}`,
    `PWD=${braced(cfg.password)}`,
    `DBQ=${cfg.library}`,
    // Second lock behind the read-only user profile: 2 = SELECT only.
    'CONNTYPE=2',
  ];
  // Only send PORT if configured; verify your driver version supports it.
  if (cfg.port) parts.push(`PORT=${cfg.port}`);
  if (cfg.ssl) parts.push('SSL=1');
  return parts.join(';') + ';';
}

// Cache the promise, not the pool. If two requests arrive before the pool is
// ready, both await the same promise instead of each creating a pool.
let poolPromise = null;
function getPool() {
  if (!poolPromise) {
    poolPromise = odbc
      .pool({
        connectionString: buildConnectionString(),
        initialSize: 2,        // warm connections: first requests avoid login cost
        incrementSize: 2,      // grow in small steps when the pool is exhausted
        maxSize: cfg.poolMax,  // hard cap: protects the AS400 from too many jobs
        connectionTimeout: 60, // seconds an idle connection may live
        loginTimeout: 15,      // seconds to wait for a login before failing
      })
      .catch((err) => {
        poolPromise = null; // allow a retry on the next call after a failure
        throw err;
      });
  }
  return poolPromise;
}

// Guard against mistakes during development. This is NOT a security boundary:
// the real barrier is the read-only user profile and CONNTYPE=2 above.
const READ_ONLY = /^\s*(select|with|values)\b/i;

async function query(sql, params = []) {
  if (!READ_ONLY.test(sql)) {
    throw new Error('Only SELECT statements are allowed');
  }
  const pool = await getPool();
  // pool.query borrows a connection, runs the statement, and returns the
  // connection to the pool. We never open/close per request (rule from AGENTS.md).
  const rows = await pool.query(sql, params);
  // CHAR columns on IBM i are space-padded; trim so callers get clean strings.
  return rows.map((row) =>
    Object.fromEntries(
      Object.entries(row).map(([k, v]) => [k, typeof v === 'string' ? v.trimEnd() : v])
    )
  );
}

async function close() {
  if (!poolPromise) return;
  const pool = await poolPromise;
  await pool.close();
  poolPromise = null;
}

module.exports = { query, close };

ทางเลือกที่พิจารณา: ใช้ connection เดี่ยวแล้วใช้ซ้ำ (ง่ายกว่า แต่ request พร้อมกันจะต่อคิว และถ้า connection หลุดต้องเขียนกู้เองทั้งหมด) จึงเลือก pool ตามกฎเดิม ส่วนค่า maxSize ต้องคุยกับ IT เพราะแต่ละ connection กินหนึ่ง job บน AS400 เริ่มต่ำ ๆ แล้วค่อยเพิ่มตามการใช้งานจริง

'use strict';
const { query, close } = require('../db');

async function main() {
  // Step A: prove connect + login. SYSDUMMY1 is a one-row system table that
  // needs no special authority, so a failure here is about connection, not data.
  const a = await query('SELECT 1 AS OK FROM SYSIBM.SYSDUMMY1');
  console.log('A. connect ok:', a[0].OK);

  // Step B: prove WHO we are and WHERE we landed. Confirms the profile IT gave us.
  const b = await query('SELECT CURRENT USER AS USR, CURRENT SERVER AS SRV FROM SYSIBM.SYSDUMMY1');
  console.log('B. user/server:', b[0].USR, b[0].SRV);

  // Step C: prove read access to one real table with a single row.
  // Replace MYLIB.MYTABLE with a real object. Keep FETCH FIRST so we never
  // scan a large production table by accident.
  const c = await query('SELECT * FROM MYLIB.MYTABLE FETCH FIRST 1 ROWS ONLY');
  console.log('C. sample columns:', Object.keys(c[0] || {}));
  // Print column names only, not values: real company data should not land in logs.
}

main()
  .catch((err) => { console.error('FAILED:', err.message); process.exitCode = 1; })
  .finally(close);

1.5 server.js: endpoint ตัวอย่างตาม pattern P_ISFOUND

server.js
'use strict';
const express = require('express');
const { query, close } = require('./db');

const app = express();

app.get('/health/db', async (req, res) => {
  try {
    const rows = await query('SELECT CURRENT USER AS USR FROM SYSIBM.SYSDUMMY1');
    res.json({ STATUS: 'OK', USR: rows[0].USR });
  } catch (err) {
    // Log the detail server-side; return a generic code to the client.
    // Driver messages can reveal host names and library names.
    console.error('health check failed:', err.message);
    res.status(503).json({ STATUS: 'DB_UNAVAILABLE' });
  }
});

app.get('/api/items/:code', async (req, res) => {
  try {
    const code = String(req.params.code);
    // Validate length before touching the database (10 = example column length).
    if (code.length === 0 || code.length > 10) {
      return res.status(400).json({ ERROR: 'INVALID_CODE' });
    }
    // XXCD / XXNM are PLACEHOLDERS. Use the real prefix from the company
    // naming table. Do not invent a prefix: if the column is not in the table, stop and ask.
    const rows = await query(
      'SELECT XXCD, XXNM FROM MYLIB.MYTABLE WHERE XXCD = ? FETCH FIRST 1 ROWS ONLY',
      [code] // bound parameter: never concatenate user input into the SQL text
    );
    // Not found is still HTTP 200 with P_ISFOUND "0" (existing contract, no 404).
    if (rows.length === 0) return res.json({ P_ISFOUND: '0' });
    res.json({ P_ISFOUND: '1', XXCD: rows[0].XXCD, XXNM: rows[0].XXNM });
  } catch (err) {
    console.error('items lookup failed:', err.message);
    res.status(500).json({ ERROR: 'INTERNAL_ERROR' });
  }
});

const server = app.listen(3000, () => console.log('listening on 3000'));

// Graceful shutdown: stop accepting requests, then close pooled connections so
// no orphan jobs stay on the AS400 after a restart.
async function shutdown() {
  server.close(async () => { await close(); process.exit(0); });
}
process.on('SIGINT', shutdown);
process.on('SIGTERM', shutdown);

1.7 Best practices ฝั่ง Node.js สรุป

  • สร้าง pool ครั้งเดียวต่อโปรเซส และเก็บเป็น promise เพื่อกันสร้างซ้ำตอน request พร้อมกัน
  • ตั้ง maxSize, loginTimeout, connectionTimeout เสมอ ไม่ปล่อยค่าเริ่มต้น
  • ทุก query ต้องมี FETCH FIRST n ROWS ONLY ตอนพัฒนา กันดึงตารางใหญ่ทั้งตาราง
  • ระบุ library นำหน้าชื่อตารางเสมอ (MYLIB.MYTABLE) ไม่พึ่ง library list เพราะเครื่อง dev กับเครื่องจริงอาจไม่เหมือนกัน
  • ข้อมูล CHAR มีช่องว่างต่อท้าย ตัดที่ชั้นแปลงข้อมูล (ตรงกับ fieldMapping ในสถาปัตยกรรมเดิม) ส่วนวันที่แบบตัวเลข (D8, D6) ให้แปลงที่ชั้นเดียวกัน
  • ไม่ส่งข้อความ error ของ driver ให้ client และไม่ log ค่า parameter
  • ปิด pool ตอน shutdown ทุกครั้ง

using System.ComponentModel.DataAnnotations;

public sealed class As400Options
{
    public const string SectionName = "As400";

    // [Required] rejects null AND empty strings, so a missing setting stops the
    // app at startup (see ValidateOnStart) instead of failing on the first request.
    [Required] public string Driver { get; set; } = "";
    [Required] public string Host { get; set; } = "";
    [Required] public string Library { get; set; } = "";
    [Required] public string User { get; set; } = "";
    [Required] public string Password { get; set; } = "";

    public int? Port { get; set; }
    public bool Ssl { get; set; }

    // Upper bound on simultaneous DB calls. Protects the AS400 (one job per
    // connection) and protects our own thread pool.
    [Range(1, 50)] public int MaxConcurrency { get; set; } = 8;
    [Range(1, 300)] public int CommandTimeoutSeconds { get; set; } = 30;
}

2.5 As400Db.cs: service สำหรับ query แบบปลอดภัย

As400Db.cs
using System.Data.Odbc;
using System.Text.RegularExpressions;
using Microsoft.Extensions.Options;

// Registered as a singleton: it holds only immutable settings and a semaphore,
// so one shared instance is safe and cheap.
public sealed partial class As400Db : IDisposable
{
    private readonly As400Options _opt;
    private readonly string _connectionString;
    private readonly SemaphoreSlim _gate;

    public As400Db(IOptions<As400Options> options)
    {
        _opt = options.Value;
        _connectionString = Build(_opt);
        _gate = new SemaphoreSlim(_opt.MaxConcurrency);
    }

    // ODBC rule: wrap in braces and double any closing brace, so a password
    // containing ';' or '}' cannot break the connection string.
    private static string Braced(string v) => "{" + v.Replace("}", "}}") + "}";

    private static string Build(As400Options o)
    {
        var parts = new List<string>
        {
            $"DRIVER={Braced(o.Driver)}",
            $"SYSTEM={o.Host}",
            $"UID={Braced(o.User)}",
            $"PWD={Braced(o.Password)}",
            $"DBQ={o.Library}",
            "CONNTYPE=2", // second lock behind the read-only profile: SELECT only
        };
        if (o.Port is int p) parts.Add($"PORT={p}"); // verify driver supports PORT
        if (o.Ssl) parts.Add("SSL=1");
        return string.Join(';', parts) + ";";
    }

    // Guard for development mistakes, not a security boundary.
    [GeneratedRegex(@"^\s*(select|with|values)\b", RegexOptions.IgnoreCase)]
    private static partial Regex ReadOnlyPattern();

    public async Task<List<Dictionary<string, object?>>> QueryAsync(
        string sql, object?[] args, CancellationToken ct)
    {
        if (!ReadOnlyPattern().IsMatch(sql))
            throw new InvalidOperationException("Only SELECT statements are allowed.");

        // Cap concurrent DB calls. System.Data.Odbc does its work synchronously
        // under the hood, so many parallel calls would occupy many threads.
        await _gate.WaitAsync(ct);
        try
        {
            // 'using' returns the connection to the driver manager pool on Dispose.
            // Opening per call is correct in ADO.NET; the pool makes it cheap.
            using var conn = new OdbcConnection(_connectionString);
            conn.Open();

            using var cmd = conn.CreateCommand();
            cmd.CommandText = sql;
            cmd.CommandTimeout = _opt.CommandTimeoutSeconds;

            // ODBC parameters are POSITIONAL: names are ignored, only order matters.
            // Each '?' in the SQL text is filled by the next parameter in this loop.
            foreach (var a in args)
                cmd.Parameters.Add(new OdbcParameter { Value = a ?? DBNull.Value });

            using var reader = cmd.ExecuteReader();
            var rows = new List<Dictionary<string, object?>>();
            while (reader.Read())
            {
                var row = new Dictionary<string, object?>(StringComparer.OrdinalIgnoreCase);
                for (var i = 0; i < reader.FieldCount; i++)
                {
                    var v = reader.GetValue(i);
                    // CHAR columns on IBM i are space-padded; trim for clean output.
                    row[reader.GetName(i)] = v is DBNull ? null : v is string s ? s.TrimEnd() : v;
                }
                rows.Add(row);
            }
            return rows;
        }
        finally
        {
            _gate.Release();
        }
    }

    public void Dispose() => _gate.Dispose();
}

2.6 Program.cs: ประกอบทุกอย่างเข้าด้วยกัน

Program.cs
using Microsoft.AspNetCore.Diagnostics;

var builder = WebApplication.CreateBuilder(args);

// Bind + validate at startup. ValidateOnStart makes a bad config crash immediately
// with a readable message instead of failing on the first database call.
builder.Services.AddOptions<As400Options>()
    .Bind(builder.Configuration.GetSection(As400Options.SectionName))
    .ValidateDataAnnotations()
    .ValidateOnStart();

builder.Services.AddSingleton<As400Db>();

var app = builder.Build();

// One place to turn any unexpected exception into a safe response.
// Detail goes to the log; the client only sees a generic code.
app.UseExceptionHandler(errorApp => errorApp.Run(async ctx =>
{
    var feature = ctx.Features.Get<IExceptionHandlerFeature>();
    app.Logger.LogError(feature?.Error, "Unhandled error on {Path}", ctx.Request.Path);
    ctx.Response.StatusCode = 500;
    await ctx.Response.WriteAsJsonAsync(new { ERROR = "INTERNAL_ERROR" });
}));

app.MapGet("/health/db", async (As400Db db, CancellationToken ct) =>
{
    var rows = await db.QueryAsync(
        "SELECT CURRENT USER AS USR, CURRENT SERVER AS SRV FROM SYSIBM.SYSDUMMY1", [], ct);
    return Results.Ok(new { STATUS = "OK", USR = rows[0]["USR"], SRV = rows[0]["SRV"] });
});

app.MapGet("/api/items/{code}", async (string code, As400Db db, CancellationToken ct) =>
{
    // Validate before touching the database (10 = example column length).
    if (code.Length is 0 or > 10)
        return Results.BadRequest(new { ERROR = "INVALID_CODE" });

    // XXCD / XXNM are PLACEHOLDERS. Use the real prefix from the company naming
    // table. Never invent a prefix: if the column is missing there, stop and ask.
    var rows = await db.QueryAsync(
        "SELECT XXCD, XXNM FROM MYLIB.MYTABLE WHERE XXCD = ? FETCH FIRST 1 ROWS ONLY",
        [code], ct);

    // Not found stays HTTP 200 with P_ISFOUND "0" (existing contract, no 404).
    return rows.Count == 0
        ? Results.Ok(new { P_ISFOUND = "0" })
        : Results.Ok(new { P_ISFOUND = "1", XXCD = rows[0]["XXCD"], XXNM = rows[0]["XXNM"] });
});

app.Run();

2.8 เรื่อง connection pooling ใน .NET (จุดที่ต่างจาก Node ที่สุด)

ต้องตรวจสอบเอง อย่าสมมติว่าเปิดอยู่

ใน System.Data.Odbc การ pool ไม่ได้ทำโดยตัวไลบรารี .NET แต่ทำโดย ODBC Driver Manager ของระบบ บน Windows ตั้งได้ที่แท็บ "Connection Pooling" ใน ODBC Administrator (ต่อ driver) บน Linux ตั้งใน odbcinst.ini ด้วยคีย์ Pooling และ CPTimeout ยืนยันรายละเอียดกับเอกสาร Microsoft และ unixODBC ของรุ่นที่ใช้

วิธีพิสูจน์ว่า pool ทำงานจริง: ให้ IT ช่วยดู job ของ database server (prestart jobs ชื่อ QZDASOINIT ใน subsystem QUSRWRK) ขณะยิง request ต่อเนื่อง ถ้า pool ทำงาน จำนวน job จะคงที่ ถ้าไม่ทำงาน จะเห็น job เกิดและดับทุก request ซึ่งขัดกับกฎ "ห้ามเปิด/ปิด connection ทุก request" ของคุณ และ semaphore ในโค้ดตัวอย่างช่วยจำกัดขอบบนให้แม้ pool จะไม่ทำงาน

2.9 Best practices ฝั่ง ASP.NET Core สรุป

  • ใช้ Options pattern พร้อม ValidateOnStart ให้ config ผิดพังตั้งแต่เริ่ม
  • secret อยู่ใน user-secrets (dev) และ environment variable (deploy) ไม่อยู่ใน appsettings.json
  • เปิด connection ด้วย using ต่อหนึ่งคำสั่ง และพึ่ง driver manager pool ห้ามเก็บ OdbcConnection ไว้เป็น field แบบ singleton
  • ODBC parameter เป็นแบบตำแหน่ง (?) เรียงลำดับให้ตรง และสร้างผ่าน OdbcParameter เสมอ
  • จำกัด concurrent ด้วย SemaphoreSlim และตั้ง CommandTimeout ทุกคำสั่ง
  • เข้าใจว่า async ของ System.Data.Odbc ส่วนใหญ่ไม่ใช่ async I/O จริง จึงไม่ควรยิงขนานจำนวนมากโดยไม่จำกัด
  • จัดการ exception ที่ middleware จุดเดียว และไม่ส่งข้อความ driver ให้ client
  • ตรวจให้แน่ว่า runtime เป็น x64 ตรงกับ driver

ส่วนที่ 5: Checklist ก่อนถือว่า "พร้อม read-only"

  • ได้ข้อมูล connection ครบจาก IT และรับรหัสผ่านผ่านช่องทางปลอดภัย
  • เครื่อง dev เข้าถึง AS400 ได้ (VPN ถ้าจำเป็น) และผ่านการทดสอบ port
  • ชื่อ driver ตรงตัวอักษรและ bitness ตรงกับ Node หรือ .NET
  • DSN ทดสอบนอกโค้ดผ่าน (ข้อ 0.5)
  • .gitignore ครอบคลุมไฟล์ secret ก่อนสร้างไฟล์นั้น
  • user profile เป็น read-only จริง และ connection string มี CONNTYPE=2
  • ทดสอบขั้น A, B, C ผ่านตามลำดับ ขั้น C อ่านแถวเดียวจากตารางจริง
  • ทุก query เป็นแบบ parameterized และมี FETCH FIRST
  • ตรวจแล้วว่า pool ทำงานจริง (จำนวน job ฝั่ง AS400 คงที่)
  • error ที่ส่งถึง client เป็นรหัสกลาง ไม่ใช่ข้อความ driver
  • ชื่อ field และ endpoint ใหม่ทุกตัวอ้างจากตาราง naming convention ของบริษัท ตัวไหนไม่มีให้หยุดถามก่อน
  • ยังไม่มี INSERT, UPDATE, DELETE, CREATE, ALTER หรือ DROP จนกว่าจะได้รับการยืนยันจากคุณเป็นลายลักษณ์อักษร
ก่อนเปิดใช้สิทธิ์เขียนในอนาคต

ต้องมีขั้นตอนอนุมัติแยกต่างหาก ใช้ profile แยกจาก profile อ่านอย่างเดียว, ทดสอบบนสำเนาข้อมูลก่อน และเปลี่ยน CONNTYPE เฉพาะ connection ที่จำเป็นเท่านั้น ไม่เปลี่ยนทั้งระบบ

ค่า port, ชื่อ driver บน Linux และคีย์เวิร์ด PORT เป็นข้อมูลที่พบโดยทั่วไป ต้องยืนยันกับ IT บริษัทและเอกสาร IBM ของรุ่นที่ติดตั้งจริง ตัวอย่าง XXCD, XXNM, MYLIB, MYTABLE เป็นตัวแทน ให้แทนด้วยชื่อจริงตามตารางมาตรฐานของบริษัท
